# UBSan suppressions for tools/ci-sanitize.sh (FP-03). Modelled on Flutter's
# engine/src/flutter/testing/ubsan_suppressions.txt. THIRD-PARTY CODE ONLY, one reason per entry.

# vptr check on libstdc++'s inline shared_ptr release (bits/shared_ptr_base.h
# _Sp_counted_base::_M_release) for control blocks ALLOCATED inside the prebuilt
# libhermesvm.so (hermes::vm::Runtime::create). Hermes is built -fno-rtti, so its
# control-block vtables carry no typeinfo the check can match. Flutter never hits this:
# it compiles -fno-rtti, where clang turns vptr off. Our own shared_ptr misuse is still
# caught by ASan (use-after-free / double free).
vptr:bits/shared_ptr_base.h

# FP-52 (props fuzzer): Yoga's StyleValuePool::isIntegerPackable does
# static_cast<int>(value) on any finite float before range-checking it, so a style
# dimension like 4e18 is a float-cast-overflow inside RN's vendored yoga. The result
# only feeds its own "fits in an int?" comparison, which then fails, and the value is
# stored as a float. Not our code; our own casts go through scenegraph/DynamicNumber.h.
float-cast-overflow:yoga/style/StyleValuePool.h

# FP-52 (image-decode fuzzer): vendored stb_image.h stbi__parse_png_file reads a
# zero-length FIRST IDAT chunk with stbi__getn(s, z->idata /* still NULL */, 0), i.e.
# memcpy(NULL, src, 0). Zero bytes are copied, so nothing is read or written, but the
# libc prototype is nonnull. Third-party code (fuzz seed image-decode/crash-stb-empty-idat).
nonnull-attribute:stb_image.h

# FP-57 (ci-prewarm under the suite): RN marks a Yoga child shared between revisions by
# setting its owner to the magic pointer 0xBADC0FFEE0DDF00D (YogaLayoutableShadowNode.cpp
# :332, documented in YogaLayoutableShadowNode.h). updateYogaChildrenOwnersIfNeeded reads it
# back through YGNodeGetOwner, whose Node* -> YGNodeRef upcast UBSan flags as misaligned.
# The pointer is only compared, never dereferenced. Third-party code, this one function.
alignment:YGNodeGetOwner
